package core

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
)

const EventSchemaVersion = 1

type SessionState string

const (
	SessionActive   SessionState = "active"
	SessionTerminal SessionState = "terminal"
	SessionDeleted  SessionState = "deleted"
)

type Session struct {
	ID         string
	Generation uint64
	Revision   uint64
	State      SessionState
	Budget     map[string]int64
}

type Event struct {
	Sequence      uint64
	SchemaVersion uint64
	SessionID     string
	Generation    uint64
	Revision      uint64
	Kind          CommandKind
	State         SessionState
	Category      string
	Amount        int64
}

type Projection struct {
	Sessions     map[string]Session
	GlobalBudget map[string]int64
	LastSequence uint64
}

func NewProjection() Projection {
	return Projection{Sessions: map[string]Session{}, GlobalBudget: map[string]int64{}}
}

func ApplyEvent(p *Projection, event Event) error {
	if event.SchemaVersion != EventSchemaVersion {
		return errors.New("unsupported event schema")
	}
	if event.Sequence <= p.LastSequence {
		return errors.New("non-monotonic event sequence")
	}
	session, exists := p.Sessions[event.SessionID]
	switch event.Kind {
	case CommandCreate:
		if exists || event.Revision != 1 {
			return errors.New("invalid create event")
		}
		session = Session{ID: event.SessionID, Generation: event.Generation, Revision: 1, State: SessionActive, Budget: map[string]int64{}}
	case CommandTransition:
		if !exists || session.State != SessionActive || event.Revision != session.Revision+1 {
			return errors.New("invalid transition event")
		}
		session.Revision = event.Revision
	case CommandReserve, CommandRelease:
		if !exists || event.Revision != session.Revision+1 || event.Category == "" || event.Amount <= 0 {
			return errors.New("invalid budget event")
		}
		delta := event.Amount
		if event.Kind == CommandRelease {
			delta = -delta
		}
		if session.Budget[event.Category]+delta < 0 || p.GlobalBudget[event.Category]+delta < 0 {
			return errors.New("budget underflow")
		}
		session.Budget[event.Category] += delta
		p.GlobalBudget[event.Category] += delta
		session.Revision = event.Revision
	case CommandTerminalize:
		if !exists || session.State != SessionActive || event.Revision != session.Revision+1 {
			return errors.New("invalid terminal event")
		}
		session.State = SessionTerminal
		session.Revision = event.Revision
	default:
		return fmt.Errorf("unknown event kind %q", event.Kind)
	}
	p.Sessions[event.SessionID] = session
	p.LastSequence = event.Sequence
	return nil
}

func EventChecksum(event Event) (string, error) {
	encoded, err := json.Marshal(struct {
		Sequence      uint64       `json:"sequence"`
		SchemaVersion uint64       `json:"schema_version"`
		SessionID     string       `json:"session_id"`
		Generation    uint64       `json:"generation"`
		Revision      uint64       `json:"revision"`
		Kind          CommandKind  `json:"kind"`
		State         SessionState `json:"state"`
		Category      string       `json:"category"`
		Amount        int64        `json:"amount"`
	}{event.Sequence, event.SchemaVersion, event.SessionID, event.Generation, event.Revision, event.Kind, event.State, event.Category, event.Amount})
	if err != nil {
		return "", err
	}
	digest := sha256.Sum256(encoded)
	return hex.EncodeToString(digest[:]), nil
}
