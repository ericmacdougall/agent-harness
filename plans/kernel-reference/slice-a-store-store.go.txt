const (
	StoreFormatVersion = 1
	databaseName       = "authority-preview.db"
	lockName           = "authority-preview.lock"
	MaxLiveSessions    = 10_000
	MaxTombstones      = 10_000
	MaxEvents          = 100_000
	MaxReceipts        = 100_000
	MaxBudgetEntries   = 100_000
	MaxSnapshotItems   = MaxLiveSessions + MaxTombstones
	MaxBudgetCategory  = int64(1_000_000_000_000)
	HardStoreBytes     = int64(4 << 30)
	EmergencyHeadroom  = int64(64 << 20)
)

var (
	ErrCommandConflict = errors.New("command-conflict")
	ErrSessionDeleted  = errors.New("session-deleted")
	ErrRevision        = errors.New("revision-conflict")
	ErrNewerStore      = errors.New("store-newer-than-binary")
	ErrInvalidState    = errors.New("invalid-state")
	ErrQuotaExceeded   = errors.New("store-quota-exceeded")
)

type Store struct {
	db                  *sql.DB
	lock                *fileLock
	dir                 string
	testBarrier         func(string)
	testSnapshotBarrier func()
}

type Receipt struct {
	CommandID     string           `json:"command_id"`
	CommandDigest string           `json:"command_digest"`
	SessionID     string           `json:"session_id"`
	Revision      uint64           `json:"revision"`
	Status        string           `json:"status"`
	Kind          core.CommandKind `json:"kind"`
}

type Tombstone struct {
	SessionID       string
	LastRevision    uint64
	TerminalState   core.SessionState
	FinalBudget     map[string]int64
	DeletionCommand string
	DeletionReceipt Receipt
	SchemaVersion   uint64
}

type Snapshot struct {
	Projection core.Projection
	Tombstones map[string]Tombstone
}

const MaxEventPage = 1000

func Open(ctx context.Context, dir string) (_ *Store, retErr error) {
	return openStore(ctx, dir, nil)
}

func openStore(ctx context.Context, dir string, barrier func(string)) (_ *Store, retErr error) {
	if dir == "" {
		return nil, errors.New("empty store directory")
	}
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return nil, err
	}
	canonicalDir, err := filepath.EvalSymlinks(dir)
	if err != nil {
		return nil, err
	}
	dir = canonicalDir
	dirInfo, err := os.Lstat(dir)
	if err != nil || !dirInfo.IsDir() || dirInfo.Mode()&os.ModeSymlink != 0 {
		return nil, errors.New("unsafe store directory")
	}
	if err := checkLocalFilesystem(dir); err != nil {
		return nil, err
	}
	lock, err := acquireLock(filepath.Join(dir, lockName))
	if err != nil {
		return nil, err
	}
	defer func() {
		if retErr != nil {
			_ = lock.close()
		}
	}()
	dbPath := filepath.Join(dir, databaseName)
	dbInfo, statErr := os.Lstat(dbPath)
	fresh := os.IsNotExist(statErr)
	if statErr != nil && !fresh {
		return nil, statErr
	}
	if !fresh {
		if !dbInfo.Mode().IsRegular() || dbInfo.Mode()&os.ModeSymlink != 0 {
			return nil, errors.New("unsafe preview database")
		}
		if err := checkDatabaseIdentity(dbInfo); err != nil {
			return nil, err
		}
	}
	if fresh {
		entries, readErr := os.ReadDir(dir)
		if readErr != nil {
			return nil, readErr
		}
		for _, entry := range entries {
			if entry.Name() != lockName {
				return nil, fmt.Errorf("preview directory is not fresh: %s", entry.Name())
			}
		}
	}
	db, err := sql.Open("sqlite", dbPath)
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(1)
	db.SetMaxIdleConns(1)
	store := &Store{db: db, lock: lock, dir: dir, testBarrier: barrier}
	initializeAsFresh := fresh
	if !fresh {
		uninitialized, err := store.isUninitialized(ctx)
		if err != nil {
			_ = db.Close()
			return nil, err
		}
		initializeAsFresh = uninitialized
	}
	if err := store.initialize(ctx, initializeAsFresh); err != nil {
		_ = db.Close()
		return nil, err
	}
	afterInfo, err := os.Lstat(dbPath)
	if err != nil || !afterInfo.Mode().IsRegular() || afterInfo.Mode()&os.ModeSymlink != 0 || (!fresh && !os.SameFile(dbInfo, afterInfo)) {
		_ = db.Close()
		return nil, errors.New("preview database identity changed")
	}
	if err := checkDatabaseIdentity(afterInfo); err != nil {
		_ = db.Close()
		return nil, err
	}
	if _, err := store.Snapshot(ctx); err != nil {
		_ = db.Close()
		return nil, fmt.Errorf("preview store integrity: %w", err)
	}
	return store, nil
}

func (s *Store) isUninitialized(ctx context.Context) (bool, error) {
	rows, err := s.db.QueryContext(ctx, `SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name`)
	if err != nil {
		return false, err
	}
	defer rows.Close()
	count := 0
	hasMetadata := false
	for rows.Next() {
		var name string
		if err := rows.Scan(&name); err != nil {
			return false, err
		}
		count++
		hasMetadata = hasMetadata || name == "metadata"
	}
	if err := rows.Err(); err != nil {
		return false, err
	}
	if count == 0 {
		return true, nil
	}
	if !hasMetadata {
		return false, errors.New("partial preview schema without metadata")
	}
	return false, nil
}

func (s *Store) initialize(ctx context.Context, fresh bool) error {
	if !fresh {
		if err := s.verifyStoreVersion(ctx); err != nil {
			return err
		}
		if err := s.verifyRecordVersions(ctx); err != nil {
			return err
		}
	}
	for _, statement := range []string{
		`PRAGMA foreign_keys=ON`,
		`PRAGMA busy_timeout=5000`,
		`PRAGMA journal_mode=WAL`,
		`PRAGMA synchronous=FULL`,
	} {
		if _, err := s.db.ExecContext(ctx, statement); err != nil {
			return err
		}
	}
	if err := s.verifyPragma(ctx, "foreign_keys", 1); err != nil {
		return err
	}
	if err := s.verifyPragma(ctx, "synchronous", 2); err != nil {
		return err
	}
	var journal string
	if err := s.db.QueryRowContext(ctx, `PRAGMA journal_mode`).Scan(&journal); err != nil || journal != "wal" {
		return fmt.Errorf("journal mode readback: %q: %w", journal, err)
	}
	if fresh {
		tx, err := s.db.BeginTx(ctx, nil)
		if err != nil {
			return err
		}
		defer tx.Rollback()
		statements := []string{
			`CREATE TABLE metadata (singleton INTEGER PRIMARY KEY CHECK(singleton=1), format_version INTEGER NOT NULL, min_writer INTEGER NOT NULL, deletion_count INTEGER NOT NULL, created_at TEXT NOT NULL)`,
			`CREATE TABLE sessions (session_id TEXT PRIMARY KEY, generation INTEGER NOT NULL, revision INTEGER NOT NULL, state TEXT NOT NULL, budget_json BLOB NOT NULL)`,
			`CREATE TABLE receipts (command_id TEXT PRIMARY KEY, session_id TEXT NOT NULL, digest TEXT NOT NULL, canonical BLOB NOT NULL, principal_json BLOB NOT NULL, receipt_json BLOB NOT NULL, checksum TEXT NOT NULL, FOREIGN KEY(session_id) REFERENCES sessions(session_id) ON DELETE CASCADE)`,
			`CREATE TABLE events (sequence INTEGER PRIMARY KEY AUTOINCREMENT, schema_version INTEGER NOT NULL, session_id TEXT NOT NULL, generation INTEGER NOT NULL, revision INTEGER NOT NULL, kind TEXT NOT NULL, state TEXT NOT NULL, category TEXT NOT NULL, amount INTEGER NOT NULL, checksum TEXT NOT NULL)`,
			`CREATE INDEX events_receipt_identity ON events(session_id,revision,kind)`,
			`CREATE TABLE global_budget (category TEXT PRIMARY KEY, amount INTEGER NOT NULL)`,
			`CREATE TABLE budget_ledger (entry_id INTEGER PRIMARY KEY AUTOINCREMENT, command_id TEXT NOT NULL UNIQUE, session_id TEXT NOT NULL, category TEXT NOT NULL, delta INTEGER NOT NULL)`,
			`CREATE TABLE tombstones (session_id TEXT PRIMARY KEY, last_revision INTEGER NOT NULL, terminal_state TEXT NOT NULL, final_budget_json BLOB NOT NULL, deletion_command_id TEXT NOT NULL UNIQUE, deletion_receipt_json BLOB NOT NULL, schema_version INTEGER NOT NULL, checksum TEXT NOT NULL)`,
		}
		for _, statement := range statements {
			if _, err := tx.ExecContext(ctx, statement); err != nil {
				return err
			}
		}
		if _, err := tx.ExecContext(ctx, `INSERT INTO metadata(singleton,format_version,min_writer,deletion_count,created_at) VALUES(1,?,?,0,?)`, StoreFormatVersion, StoreFormatVersion, time.Now().UTC().Format(time.RFC3339Nano)); err != nil {
			return err
		}
		if s.testBarrier != nil {
			s.testBarrier("initialization-before-commit")
		}
		if err := tx.Commit(); err != nil {
			return err
		}
		if s.testBarrier != nil {
			s.testBarrier("initialization-after-commit")
		}
	}
	if err := s.verifyStoreVersion(ctx); err != nil {
		return err
	}
	return s.verifyRecordVersions(ctx)
}

func (s *Store) verifyStoreVersion(ctx context.Context) error {
	var format, minWriter uint64
	if err := s.db.QueryRowContext(ctx, `SELECT format_version,min_writer FROM metadata WHERE singleton=1`).Scan(&format, &minWriter); err != nil {
		return err
	}
	if format > StoreFormatVersion || minWriter > StoreFormatVersion {
		return ErrNewerStore
	}
	return nil
}

func (s *Store) verifyRecordVersions(ctx context.Context) error {
	var unsupported int
	if err := s.db.QueryRowContext(ctx, `SELECT COUNT(*) FROM events WHERE schema_version>?`, core.EventSchemaVersion).Scan(&unsupported); err != nil {
		return err
	}
	if unsupported != 0 {
		return ErrNewerStore
	}
	if err := s.db.QueryRowContext(ctx, `SELECT COUNT(*) FROM tombstones WHERE schema_version>?`, core.EventSchemaVersion).Scan(&unsupported); err != nil {
		return err
	}
	if unsupported != 0 {
		return ErrNewerStore
	}
	return nil
}

func (s *Store) verifyPragma(ctx context.Context, name string, want int) error {
	var got int
	if err := s.db.QueryRowContext(ctx, `PRAGMA `+name).Scan(&got); err != nil {
		return err
	}
	if got != want {
		return fmt.Errorf("pragma %s=%d, want %d", name, got, want)
	}
	return nil
}

func (s *Store) Close() error {
	if s == nil {
		return nil
	}
	var dbErr error
	if s.db != nil {
		dbErr = s.db.Close()
	}
	lockErr := s.lock.close()
	if dbErr != nil {
		return dbErr
	}
	return lockErr
}

func (s *Store) Submit(ctx context.Context, principal core.Principal, raw []byte) (Receipt, error) {
	cmd, canonical, _, err := core.DecodeCommand(raw)
	if err != nil {
		return Receipt{}, err
	}
	if err := core.Authorize(principal, cmd); err != nil {
		return Receipt{}, err
	}
	digest, err := core.BindPrincipalDigest(canonical, principal)
	if err != nil {
		return Receipt{}, err
	}
	if cmd.Kind != core.CommandDelete {
		if err := s.checkStorageHeadroom(); err != nil {
			return Receipt{}, err
		}
	}
	if s.testBarrier != nil {
		s.testBarrier("before-transaction")
	}
	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return Receipt{}, err
	}
	defer tx.Rollback()
	if err := validateTombstoneCount(ctx, tx); err != nil {
		return Receipt{}, err
	}

	if tombstone, found, err := readTombstone(ctx, tx, cmd.SessionID); err != nil {
		return Receipt{}, err
	} else if found {
		if cmd.ID == tombstone.DeletionCommand {
			if tombstone.DeletionReceipt.CommandDigest != digest {
				return Receipt{}, ErrCommandConflict
			}
			return tombstone.DeletionReceipt, tx.Commit()
		}
		return Receipt{}, ErrSessionDeleted
	}
	if tombstone, found, err := readTombstoneByCommand(ctx, tx, cmd.ID); err != nil {
		return Receipt{}, err
	} else if found {
		if tombstone.SessionID != cmd.SessionID || tombstone.DeletionReceipt.CommandDigest != digest {
			return Receipt{}, ErrCommandConflict
		}
		return tombstone.DeletionReceipt, tx.Commit()
	}
	if receipt, existingDigest, found, err := readReceipt(ctx, tx, cmd.ID); err != nil {
		return Receipt{}, err
	} else if found {
		if existingDigest != digest {
			return Receipt{}, ErrCommandConflict
		}
		return receipt, tx.Commit()
	}
	receipt, err := s.apply(ctx, tx, principal, cmd, canonical, digest)
	if err != nil {
		return Receipt{}, err
	}
	if s.testBarrier != nil {
		s.testBarrier("before-commit")
	}
	if err := tx.Commit(); err != nil {
		return Receipt{}, err
	}
	if s.testBarrier != nil {
		s.testBarrier("after-commit")
	}
	return receipt, nil
}

func (s *Store) apply(ctx context.Context, tx *sql.Tx, principal core.Principal, cmd core.Command, canonical []byte, digest string) (Receipt, error) {
	session, exists, err := readSession(ctx, tx, cmd.SessionID)
	if err != nil {
		return Receipt{}, err
	}
	if cmd.Kind == core.CommandCreate {
		if exists || cmd.ExpectedRevision != 0 || cmd.Generation == 0 {
			return Receipt{}, ErrRevision
		}
		session = core.Session{ID: cmd.SessionID, Generation: cmd.Generation, Revision: 1, State: core.SessionActive, Budget: map[string]int64{}}
		if err := enforceCountLimit(ctx, tx, `SELECT COUNT(*) FROM sessions`, MaxLiveSessions); err != nil {
			return Receipt{}, err
		}
	} else {
		if !exists || cmd.ExpectedRevision != session.Revision || cmd.Generation != session.Generation {
			return Receipt{}, ErrRevision
		}
		if session.State != core.SessionActive && cmd.Kind != core.CommandDelete {
			return Receipt{}, ErrInvalidState
		}
		session.Revision++
	}

	if cmd.Kind == core.CommandDelete {
		if session.State != core.SessionTerminal {
			return Receipt{}, ErrInvalidState
		}
		if err := enforceCountLimit(ctx, tx, `SELECT COUNT(*) FROM tombstones`, MaxTombstones); err != nil {
			return Receipt{}, err
		}
		receipt := Receipt{CommandID: cmd.ID, CommandDigest: digest, SessionID: cmd.SessionID, Revision: session.Revision, Status: "accepted", Kind: cmd.Kind}
		budgetJSON, _ := json.Marshal(session.Budget)
		receiptJSON, _ := json.Marshal(receipt)
		tombstoneChecksum := checksumFields([]byte(session.ID), uint64Bytes(session.Revision), []byte(core.SessionTerminal), budgetJSON, []byte(cmd.ID), receiptJSON, uint64Bytes(core.EventSchemaVersion))
		if _, err := tx.ExecContext(ctx, `INSERT INTO tombstones(session_id,last_revision,terminal_state,final_budget_json,deletion_command_id,deletion_receipt_json,schema_version,checksum) VALUES(?,?,?,?,?,?,?,?)`, session.ID, session.Revision, core.SessionTerminal, budgetJSON, cmd.ID, receiptJSON, core.EventSchemaVersion, tombstoneChecksum); err != nil {
			return Receipt{}, err
		}
		if _, err := tx.ExecContext(ctx, `DELETE FROM receipts WHERE session_id=?`, session.ID); err != nil {
			return Receipt{}, err
		}
		if _, err := tx.ExecContext(ctx, `DELETE FROM events WHERE session_id=?`, session.ID); err != nil {
			return Receipt{}, err
		}
		if _, err := tx.ExecContext(ctx, `DELETE FROM sessions WHERE session_id=?`, session.ID); err != nil {
			return Receipt{}, err
		}
		if _, err := tx.ExecContext(ctx, `UPDATE metadata SET deletion_count=deletion_count+1 WHERE singleton=1`); err != nil {
			return Receipt{}, err
		}
		return receipt, nil
	}

	switch cmd.Kind {
	case core.CommandCreate:
	case core.CommandTransition:
		if cmd.Payload.State == "" {
			return Receipt{}, ErrInvalidState
		}
	case core.CommandReserve, core.CommandRelease:
		if !core.IsBudgetCategory(cmd.Payload.Category) || cmd.Payload.Amount <= 0 || cmd.Payload.Amount > MaxBudgetCategory {
			return Receipt{}, ErrInvalidState
		}
		delta := cmd.Payload.Amount
		if cmd.Kind == core.CommandRelease {
			delta = -delta
		}
		nextSession, ok := checkedAdd(session.Budget[cmd.Payload.Category], delta)
		if !ok || nextSession < 0 || nextSession > MaxBudgetCategory {
			return Receipt{}, ErrInvalidState
		}
		var global int64
		err := tx.QueryRowContext(ctx, `SELECT amount FROM global_budget WHERE category=?`, cmd.Payload.Category).Scan(&global)
		if err != nil && !errors.Is(err, sql.ErrNoRows) {
			return Receipt{}, err
		}
		nextGlobal, ok := checkedAdd(global, delta)
		if !ok || nextGlobal < 0 || nextGlobal > MaxBudgetCategory {
			return Receipt{}, ErrQuotaExceeded
		}
		session.Budget[cmd.Payload.Category] = nextSession
		if _, err := tx.ExecContext(ctx, `INSERT INTO global_budget(category,amount) VALUES(?,?) ON CONFLICT(category) DO UPDATE SET amount=excluded.amount`, cmd.Payload.Category, nextGlobal); err != nil {
			return Receipt{}, err
		}
		if err := enforceCountLimit(ctx, tx, `SELECT COUNT(*) FROM budget_ledger`, MaxBudgetEntries); err != nil {
			return Receipt{}, err
		}
		if _, err := tx.ExecContext(ctx, `INSERT INTO budget_ledger(command_id,session_id,category,delta) VALUES(?,?,?,?)`, cmd.ID, cmd.SessionID, cmd.Payload.Category, delta); err != nil {
			return Receipt{}, err
		}
	case core.CommandTerminalize:
		session.State = core.SessionTerminal
	default:
		return Receipt{}, ErrInvalidState
	}

	budgetJSON, _ := json.Marshal(session.Budget)
	if cmd.Kind == core.CommandCreate {
		_, err = tx.ExecContext(ctx, `INSERT INTO sessions(session_id,generation,revision,state,budget_json) VALUES(?,?,?,?,?)`, session.ID, session.Generation, session.Revision, session.State, budgetJSON)
	} else {
		_, err = tx.ExecContext(ctx, `UPDATE sessions SET revision=?,state=?,budget_json=? WHERE session_id=?`, session.Revision, session.State, budgetJSON, session.ID)
	}
	if err != nil {
		return Receipt{}, err
	}
	if err := enforceCountLimit(ctx, tx, `SELECT COUNT(*) FROM events`, MaxEvents); err != nil {
		return Receipt{}, err
	}
	result, err := tx.ExecContext(ctx, `INSERT INTO events(schema_version,session_id,generation,revision,kind,state,category,amount,checksum) VALUES(?,?,?,?,?,?,?,?,?)`, core.EventSchemaVersion, session.ID, session.Generation, session.Revision, cmd.Kind, session.State, cmd.Payload.Category, cmd.Payload.Amount, "")
	if err != nil {
		return Receipt{}, err
	}
	sequence, err := result.LastInsertId()
	if err != nil {
		return Receipt{}, err
	}
	event := core.Event{Sequence: uint64(sequence), SchemaVersion: core.EventSchemaVersion, SessionID: session.ID, Generation: session.Generation, Revision: session.Revision, Kind: cmd.Kind, State: session.State, Category: cmd.Payload.Category, Amount: cmd.Payload.Amount}
	checksum, err := core.EventChecksum(event)
	if err != nil {
		return Receipt{}, err
	}
	if _, err := tx.ExecContext(ctx, `UPDATE events SET checksum=? WHERE sequence=?`, checksum, sequence); err != nil {
		return Receipt{}, err
	}
	receipt := Receipt{CommandID: cmd.ID, CommandDigest: digest, SessionID: cmd.SessionID, Revision: session.Revision, Status: "accepted", Kind: cmd.Kind}
	receiptJSON, _ := json.Marshal(receipt)
	principalJSON, err := core.CanonicalPrincipal(principal)
	if err != nil {
		return Receipt{}, err
	}
	receiptChecksum := checksumFields([]byte(cmd.ID), []byte(cmd.SessionID), []byte(digest), canonical, principalJSON, receiptJSON)
	if err := enforceCountLimit(ctx, tx, `SELECT COUNT(*) FROM receipts`, MaxReceipts); err != nil {
		return Receipt{}, err
	}
	if _, err := tx.ExecContext(ctx, `INSERT INTO receipts(command_id,session_id,digest,canonical,principal_json,receipt_json,checksum) VALUES(?,?,?,?,?,?,?)`, cmd.ID, cmd.SessionID, digest, canonical, principalJSON, receiptJSON, receiptChecksum); err != nil {
		return Receipt{}, err
	}
	return receipt, nil
}

func readReceipt(ctx context.Context, tx *sql.Tx, id string) (Receipt, string, bool, error) {
	var sessionID, digest, checksum string
	var canonical, principalJSON, receiptJSON []byte
	var eventCount int
	err := tx.QueryRowContext(ctx, `SELECT r.session_id,r.digest,r.canonical,r.principal_json,r.receipt_json,r.checksum,(SELECT COUNT(*) FROM events e WHERE e.session_id=r.session_id AND e.revision=json_extract(r.receipt_json,'$.revision') AND e.kind=json_extract(r.receipt_json,'$.kind')) FROM receipts r WHERE r.command_id=?`, id).Scan(&sessionID, &digest, &canonical, &principalJSON, &receiptJSON, &checksum, &eventCount)
	if errors.Is(err, sql.ErrNoRows) {
		return Receipt{}, "", false, nil
	}
	if err != nil {
		return Receipt{}, "", false, err
	}
	receipt, err := validateReceiptRow(id, sessionID, digest, canonical, principalJSON, receiptJSON, checksum, eventCount)
	if err != nil {
		return Receipt{}, "", false, err
	}
	return receipt, digest, true, nil
}

func validateReceiptRow(commandID, sessionID, digest string, canonical, principalJSON, receiptJSON []byte, checksum string, eventCount int) (Receipt, error) {
	if checksum != checksumFields([]byte(commandID), []byte(sessionID), []byte(digest), canonical, principalJSON, receiptJSON) {
		return Receipt{}, errors.New("receipt checksum mismatch")
	}
	cmd, normalized, _, err := core.DecodeCommand(canonical)
	if err != nil || !reflect.DeepEqual(normalized, canonical) || cmd.ID != commandID || cmd.SessionID != sessionID {
		return Receipt{}, errors.New("receipt canonical command mismatch")
	}
	var principal core.Principal
	if err := json.Unmarshal(principalJSON, &principal); err != nil {
		return Receipt{}, err
	}
	if err := core.Authorize(principal, cmd); err != nil {
		return Receipt{}, errors.New("receipt principal is not authorized")
	}
	boundDigest, err := core.BindPrincipalDigest(canonical, principal)
	if err != nil || boundDigest != digest {
		return Receipt{}, errors.New("receipt principal digest mismatch")
	}
	var receipt Receipt
	if err := json.Unmarshal(receiptJSON, &receipt); err != nil {
		return Receipt{}, err
	}
	if receipt.CommandID != commandID || receipt.CommandDigest != digest || receipt.SessionID != sessionID || receipt.Revision == 0 || receipt.Status != "accepted" || receipt.Kind != cmd.Kind {
		return Receipt{}, errors.New("receipt fields mismatch")
	}
	if eventCount != 1 {
		return Receipt{}, errors.New("receipt event identity mismatch")
	}
	return receipt, nil
}

func readSession(ctx context.Context, tx *sql.Tx, id string) (core.Session, bool, error) {
	var session core.Session
	var state string
	var raw []byte
	err := tx.QueryRowContext(ctx, `SELECT session_id,generation,revision,state,budget_json FROM sessions WHERE session_id=?`, id).Scan(&session.ID, &session.Generation, &session.Revision, &state, &raw)
	if errors.Is(err, sql.ErrNoRows) {
		return core.Session{}, false, nil
	}
	if err != nil {
		return core.Session{}, false, err
	}
	session.State = core.SessionState(state)
	if err := json.Unmarshal(raw, &session.Budget); err != nil {
		return core.Session{}, false, err
	}
	return session, true, nil
}

func readTombstone(ctx context.Context, tx *sql.Tx, id string) (Tombstone, bool, error) {
	var tombstone Tombstone
	var state, checksum string
	var budgetJSON, receiptJSON []byte
	err := tx.QueryRowContext(ctx, `SELECT session_id,last_revision,terminal_state,final_budget_json,deletion_command_id,deletion_receipt_json,schema_version,checksum FROM tombstones WHERE session_id=?`, id).Scan(&tombstone.SessionID, &tombstone.LastRevision, &state, &budgetJSON, &tombstone.DeletionCommand, &receiptJSON, &tombstone.SchemaVersion, &checksum)
	if errors.Is(err, sql.ErrNoRows) {
		return Tombstone{}, false, nil
	}
	if err != nil {
		return Tombstone{}, false, err
	}
	tombstone.TerminalState = core.SessionState(state)
	if checksum != checksumFields([]byte(tombstone.SessionID), uint64Bytes(tombstone.LastRevision), []byte(state), budgetJSON, []byte(tombstone.DeletionCommand), receiptJSON, uint64Bytes(tombstone.SchemaVersion)) {
		return Tombstone{}, false, errors.New("tombstone checksum mismatch")
	}
	if tombstone.SchemaVersion != core.EventSchemaVersion {
		return Tombstone{}, false, ErrNewerStore
	}
	if err := json.Unmarshal(budgetJSON, &tombstone.FinalBudget); err != nil {
		return Tombstone{}, false, err
	}
	if err := json.Unmarshal(receiptJSON, &tombstone.DeletionReceipt); err != nil {
		return Tombstone{}, false, err
	}
	if tombstone.TerminalState != core.SessionTerminal || tombstone.LastRevision == 0 || tombstone.DeletionReceipt.CommandID != tombstone.DeletionCommand || tombstone.DeletionReceipt.SessionID != tombstone.SessionID || tombstone.DeletionReceipt.Revision != tombstone.LastRevision || tombstone.DeletionReceipt.Status != "accepted" || tombstone.DeletionReceipt.Kind != core.CommandDelete || len(tombstone.DeletionReceipt.CommandDigest) != 64 {
		return Tombstone{}, false, errors.New("invalid deletion tombstone receipt")
	}
	_, bySession, err := readBudgetLedger(ctx, tx)
	if err != nil {
		return Tombstone{}, false, err
	}
	if !budgetMapsEqual(tombstone.FinalBudget, bySession[tombstone.SessionID]) {
		return Tombstone{}, false, errors.New("tombstone budget disagrees with append-only ledger")
	}
	return tombstone, true, nil
}

func readTombstoneByCommand(ctx context.Context, tx *sql.Tx, commandID string) (Tombstone, bool, error) {
	var sessionID string
	err := tx.QueryRowContext(ctx, `SELECT session_id FROM tombstones WHERE deletion_command_id=?`, commandID).Scan(&sessionID)
	if errors.Is(err, sql.ErrNoRows) {
		return Tombstone{}, false, nil
	}
	if err != nil {
		return Tombstone{}, false, err
	}
	return readTombstone(ctx, tx, sessionID)
}

func enforceCountLimit(ctx context.Context, tx *sql.Tx, query string, limit int) error {
	var count int
	if err := tx.QueryRowContext(ctx, query).Scan(&count); err != nil {
		return err
	}
	if count >= limit {
		return ErrQuotaExceeded
	}
	return nil
}

func checkedAdd(left, right int64) (int64, bool) {
	if right > 0 && left > math.MaxInt64-right {
		return 0, false
	}
	if right < 0 && left < math.MinInt64-right {
		return 0, false
	}
	return left + right, true
}

func (s *Store) checkStorageHeadroom() error {
	var total int64
	for _, suffix := range []string{"", "-wal", "-shm"} {
		info, err := os.Stat(filepath.Join(s.dir, databaseName) + suffix)
		if err != nil {
			if os.IsNotExist(err) {
				continue
			}
			return err
		}
		total += info.Size()
	}
	if exceedsStorageQuota(total) {
		return ErrQuotaExceeded
	}
	return nil
}

func exceedsStorageQuota(total int64) bool {
	return total > HardStoreBytes-EmergencyHeadroom
}

func (s *Store) Snapshot(ctx context.Context) (Snapshot, error) {
	tx, err := s.db.BeginTx(ctx, &sql.TxOptions{ReadOnly: true})
	if err != nil {
		return Snapshot{}, err
	}
	defer tx.Rollback()
	for _, check := range []struct {
		query string
		limit int
	}{
		{`SELECT COUNT(*) FROM sessions`, MaxLiveSessions},
		{`SELECT COUNT(*) FROM tombstones`, MaxTombstones},
		{`SELECT COUNT(*) FROM events`, MaxEvents},
		{`SELECT COUNT(*) FROM receipts`, MaxReceipts},
		{`SELECT COUNT(*) FROM budget_ledger`, MaxBudgetEntries},
	} {
		var count int
		if err := tx.QueryRowContext(ctx, check.query).Scan(&count); err != nil {
			return Snapshot{}, err
		}
		if count > check.limit {
			return Snapshot{}, ErrQuotaExceeded
		}
	}
	var snapshotItems int
	if err := tx.QueryRowContext(ctx, `SELECT (SELECT COUNT(*) FROM sessions)+(SELECT COUNT(*) FROM tombstones)`).Scan(&snapshotItems); err != nil {
		return Snapshot{}, err
	}
	if snapshotItems > MaxSnapshotItems {
		return Snapshot{}, ErrQuotaExceeded
	}
	projection := core.NewProjection()
	rows, err := tx.QueryContext(ctx, `SELECT sequence,schema_version,session_id,generation,revision,kind,state,category,amount,checksum FROM events ORDER BY sequence`)
	if err != nil {
		return Snapshot{}, err
	}
	for rows.Next() {
		var event core.Event
		var kind, state string
		var storedChecksum string
		if err := rows.Scan(&event.Sequence, &event.SchemaVersion, &event.SessionID, &event.Generation, &event.Revision, &kind, &state, &event.Category, &event.Amount, &storedChecksum); err != nil {
			rows.Close()
			return Snapshot{}, err
		}
		event.Kind = core.CommandKind(kind)
		event.State = core.SessionState(state)
		checksum, err := core.EventChecksum(event)
		if err != nil || checksum != storedChecksum {
			rows.Close()
			return Snapshot{}, errors.New("event checksum mismatch")
		}
		if err := core.ApplyEvent(&projection, event); err != nil {
			rows.Close()
			return Snapshot{}, err
		}
	}
	if err := rows.Close(); err != nil {
		return Snapshot{}, err
	}
	if err := rows.Err(); err != nil {
		return Snapshot{}, err
	}
	if s.testSnapshotBarrier != nil {
		s.testSnapshotBarrier()
	}
	var durableSequence uint64
	if err := tx.QueryRowContext(ctx, `SELECT COALESCE(seq,0) FROM sqlite_sequence WHERE name='events'`).Scan(&durableSequence); err != nil && !errors.Is(err, sql.ErrNoRows) {
		return Snapshot{}, err
	}
	if durableSequence > projection.LastSequence {
		projection.LastSequence = durableSequence
	}
	materialized, err := readMaterializedSessions(ctx, tx)
	if err != nil {
		return Snapshot{}, err
	}
	if !reflect.DeepEqual(materialized, projection.Sessions) {
		return Snapshot{}, errors.New("materialized sessions disagree with event replay")
	}
	if err := validateReceipts(ctx, tx); err != nil {
		return Snapshot{}, err
	}

	projection.GlobalBudget = map[string]int64{}
	budgetRows, err := tx.QueryContext(ctx, `SELECT category,amount FROM global_budget ORDER BY category`)
	if err != nil {
		return Snapshot{}, err
	}
	for budgetRows.Next() {
		var category string
		var amount int64
		if err := budgetRows.Scan(&category, &amount); err != nil {
			budgetRows.Close()
			return Snapshot{}, err
		}
		if !core.IsBudgetCategory(category) || amount < 0 || amount > MaxBudgetCategory {
			budgetRows.Close()
			return Snapshot{}, errors.New("invalid global budget row")
		}
		projection.GlobalBudget[category] = amount
	}
	if err := budgetRows.Close(); err != nil {
		return Snapshot{}, err
	}
	ledgerGlobal, ledgerBySession, err := readBudgetLedger(ctx, tx)
	if err != nil {
		return Snapshot{}, err
	}
	if !budgetMapsEqual(ledgerGlobal, projection.GlobalBudget) {
		return Snapshot{}, errors.New("global budget disagrees with append-only ledger")
	}
	for sessionID, session := range projection.Sessions {
		if !budgetMapsEqual(session.Budget, ledgerBySession[sessionID]) {
			return Snapshot{}, errors.New("session budget disagrees with append-only ledger")
		}
	}

	tombstones := map[string]Tombstone{}
	tombstoneRows, err := tx.QueryContext(ctx, `SELECT session_id,last_revision,terminal_state,final_budget_json,deletion_command_id,deletion_receipt_json,schema_version,checksum FROM tombstones ORDER BY session_id`)
	if err != nil {
		return Snapshot{}, err
	}
	for tombstoneRows.Next() {
		var tombstone Tombstone
		var state, checksum string
		var budgetJSON, receiptJSON []byte
		if err := tombstoneRows.Scan(&tombstone.SessionID, &tombstone.LastRevision, &state, &budgetJSON, &tombstone.DeletionCommand, &receiptJSON, &tombstone.SchemaVersion, &checksum); err != nil {
			tombstoneRows.Close()
			return Snapshot{}, err
		}
		if tombstone.SchemaVersion != core.EventSchemaVersion {
			tombstoneRows.Close()
			return Snapshot{}, ErrNewerStore
		}
		tombstone.TerminalState = core.SessionState(state)
		if checksum != checksumFields([]byte(tombstone.SessionID), uint64Bytes(tombstone.LastRevision), []byte(state), budgetJSON, []byte(tombstone.DeletionCommand), receiptJSON, uint64Bytes(tombstone.SchemaVersion)) {
			tombstoneRows.Close()
			return Snapshot{}, errors.New("tombstone checksum mismatch")
		}
		if err := json.Unmarshal(budgetJSON, &tombstone.FinalBudget); err != nil {
			tombstoneRows.Close()
			return Snapshot{}, err
		}
		if err := json.Unmarshal(receiptJSON, &tombstone.DeletionReceipt); err != nil {
			tombstoneRows.Close()
			return Snapshot{}, err
		}
		if tombstone.TerminalState != core.SessionTerminal || tombstone.LastRevision == 0 || tombstone.DeletionReceipt.CommandID != tombstone.DeletionCommand || tombstone.DeletionReceipt.SessionID != tombstone.SessionID || tombstone.DeletionReceipt.Revision != tombstone.LastRevision || tombstone.DeletionReceipt.Status != "accepted" || tombstone.DeletionReceipt.Kind != core.CommandDelete || len(tombstone.DeletionReceipt.CommandDigest) != 64 {
			tombstoneRows.Close()
			return Snapshot{}, errors.New("invalid deletion tombstone receipt")
		}
		if !budgetMapsEqual(tombstone.FinalBudget, ledgerBySession[tombstone.SessionID]) {
			tombstoneRows.Close()
			return Snapshot{}, errors.New("tombstone budget disagrees with append-only ledger")
		}
		tombstones[tombstone.SessionID] = tombstone
		projection.Sessions[tombstone.SessionID] = core.Session{ID: tombstone.SessionID, Revision: tombstone.LastRevision, State: core.SessionDeleted, Budget: cloneBudget(tombstone.FinalBudget)}
	}
	if err := tombstoneRows.Close(); err != nil {
		return Snapshot{}, err
	}
	var deletionCount uint64
	if err := tx.QueryRowContext(ctx, `SELECT deletion_count FROM metadata WHERE singleton=1`).Scan(&deletionCount); err != nil {
		return Snapshot{}, err
	}
	if deletionCount != uint64(len(tombstones)) {
		return Snapshot{}, errors.New("deleted session tombstone is missing")
	}
	if err := tx.Commit(); err != nil {
		return Snapshot{}, err
	}
	return Snapshot{Projection: projection, Tombstones: tombstones}, nil
}

func (s *Store) ReadEvents(ctx context.Context, after uint64, limit int) ([]core.Event, error) {
	if limit <= 0 || limit > MaxEventPage {
		return nil, errors.New("event page limit is out of range")
	}
	rows, err := s.db.QueryContext(ctx, `SELECT sequence,schema_version,session_id,generation,revision,kind,state,category,amount,checksum FROM events WHERE sequence>? ORDER BY sequence LIMIT ?`, after, limit)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	events := make([]core.Event, 0, limit)
	for rows.Next() {
		var event core.Event
		var kind, state string
		var storedChecksum string
		if err := rows.Scan(&event.Sequence, &event.SchemaVersion, &event.SessionID, &event.Generation, &event.Revision, &kind, &state, &event.Category, &event.Amount, &storedChecksum); err != nil {
			return nil, err
		}
		if event.SchemaVersion != core.EventSchemaVersion {
			return nil, ErrNewerStore
		}
		event.Kind = core.CommandKind(kind)
		event.State = core.SessionState(state)
		checksum, err := core.EventChecksum(event)
		if err != nil || checksum != storedChecksum {
			return nil, errors.New("event checksum mismatch")
		}
		events = append(events, event)
	}
	return events, rows.Err()
}

func cloneBudget(input map[string]int64) map[string]int64 {
	result := make(map[string]int64, len(input))
	for key, value := range input {
		result[key] = value
	}
	return result
}

func validateTombstoneCount(ctx context.Context, tx *sql.Tx) error {
	var expected, actual int
	if err := tx.QueryRowContext(ctx, `SELECT deletion_count FROM metadata WHERE singleton=1`).Scan(&expected); err != nil {
		return err
	}
	if err := tx.QueryRowContext(ctx, `SELECT COUNT(*) FROM tombstones`).Scan(&actual); err != nil {
		return err
	}
	if expected != actual {
		return errors.New("deleted session tombstone is missing")
	}
	return nil
}

func readMaterializedSessions(ctx context.Context, tx *sql.Tx) (map[string]core.Session, error) {
	rows, err := tx.QueryContext(ctx, `SELECT session_id,generation,revision,state,budget_json FROM sessions ORDER BY session_id`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	result := map[string]core.Session{}
	for rows.Next() {
		var session core.Session
		var state string
		var raw []byte
		if err := rows.Scan(&session.ID, &session.Generation, &session.Revision, &state, &raw); err != nil {
			return nil, err
		}
		session.State = core.SessionState(state)
		if session.State != core.SessionActive && session.State != core.SessionTerminal {
			return nil, errors.New("invalid materialized session state")
		}
		if err := json.Unmarshal(raw, &session.Budget); err != nil {
			return nil, err
		}
		result[session.ID] = session
	}
	return result, rows.Err()
}

func readBudgetLedger(ctx context.Context, tx *sql.Tx) (map[string]int64, map[string]map[string]int64, error) {
	rows, err := tx.QueryContext(ctx, `SELECT session_id,category,delta FROM budget_ledger ORDER BY entry_id`)
	if err != nil {
		return nil, nil, err
	}
	defer rows.Close()
	global := map[string]int64{}
	bySession := map[string]map[string]int64{}
	for rows.Next() {
		var sessionID, category string
		var delta int64
		if err := rows.Scan(&sessionID, &category, &delta); err != nil {
			return nil, nil, err
		}
		if !core.IsBudgetCategory(category) || delta == 0 {
			return nil, nil, errors.New("invalid budget ledger row")
		}
		if bySession[sessionID] == nil {
			bySession[sessionID] = map[string]int64{}
		}
		nextSession, ok := checkedAdd(bySession[sessionID][category], delta)
		if !ok || nextSession < 0 || nextSession > MaxBudgetCategory {
			return nil, nil, errors.New("session budget ledger overflow")
		}
		nextGlobal, ok := checkedAdd(global[category], delta)
		if !ok || nextGlobal < 0 || nextGlobal > MaxBudgetCategory {
			return nil, nil, errors.New("global budget ledger overflow")
		}
		bySession[sessionID][category] = nextSession
		global[category] = nextGlobal
	}
	if err := rows.Err(); err != nil {
		return nil, nil, err
	}
	return global, bySession, nil
}

func budgetMapsEqual(left, right map[string]int64) bool {
	for category, amount := range left {
		if amount != 0 && right[category] != amount {
			return false
		}
		if amount == 0 {
			if other, exists := right[category]; exists && other != 0 {
				return false
			}
		}
	}
	for category, amount := range right {
		if amount != 0 && left[category] != amount {
			return false
		}
	}
	return true
}

func validateReceipts(ctx context.Context, tx *sql.Tx) error {
	var oversized int
	if err := tx.QueryRowContext(ctx, `SELECT COUNT(*) FROM receipts WHERE length(canonical)>? OR length(principal_json)>1024 OR length(receipt_json)>2048 OR length(checksum)!=64 OR length(digest)!=64`, core.MaxCommandBytes).Scan(&oversized); err != nil {
		return err
	}
	if oversized != 0 {
		return errors.New("receipt field exceeds integrity bound")
	}
	rows, err := tx.QueryContext(ctx, `SELECT r.command_id,r.session_id,r.digest,r.canonical,r.principal_json,r.receipt_json,r.checksum,(SELECT COUNT(*) FROM events e WHERE e.session_id=r.session_id AND e.revision=json_extract(r.receipt_json,'$.revision') AND e.kind=json_extract(r.receipt_json,'$.kind')) FROM receipts r ORDER BY r.command_id`)
	if err != nil {
		return err
	}
	defer rows.Close()
	for rows.Next() {
		var commandID, sessionID, digest, checksum string
		var canonical, principal, receipt []byte
		var eventCount int
		if err := rows.Scan(&commandID, &sessionID, &digest, &canonical, &principal, &receipt, &checksum, &eventCount); err != nil {
			return err
		}
		if _, err := validateReceiptRow(commandID, sessionID, digest, canonical, principal, receipt, checksum, eventCount); err != nil {
			return err
		}
	}
	return rows.Err()
}

func checksumFields(fields ...[]byte) string {
	digest := sha256.New()
	var size [8]byte
	for _, field := range fields {
		binary.BigEndian.PutUint64(size[:], uint64(len(field)))
		_, _ = digest.Write(size[:])
		_, _ = digest.Write(field)
	}
	return fmt.Sprintf("%x", digest.Sum(nil))
}

func uint64Bytes(value uint64) []byte {
	result := make([]byte, 8)
	binary.BigEndian.PutUint64(result, value)
	return result
}
